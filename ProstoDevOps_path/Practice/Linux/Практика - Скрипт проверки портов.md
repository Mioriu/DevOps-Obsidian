cat check_ports.sh && echo "---" && ./check_ports.sh

#!/bin/bash
HOSTS="google.com:443 google.com:80 localhost:22 localhost:9999"
for i in $HOSTS; do
 HOST=$(echo $i | cut -d: -f1)
 PORT=$(echo $i | cut -d: -f2)
 if nc -zv $HOST $PORT > /dev/null 2>&1; then
    echo "$i - OK"
 else
    echo "$i - FAIL"
 fi
done
---
google.com:443 - OK
google.com:80 - OK
localhost:22 - OK
localhost:9999 - FAIL
